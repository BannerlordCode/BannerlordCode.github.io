current_version = path_parts | nth(n=1) | default(value="") -%} {%- if nav.meta.versions is containing(current_version) -%} {%- set current_lang = path_parts | nth(n=2) | default(value=lang) -%} {%- if not (nav.meta.languages is containing(current_lang)) -%} {%- set current_lang = lang -%} {%- endif -%} {%- set root_key = "/" ~ current_version ~ "/" ~ current_lang ~ "/" -%} {%- else -%} {%- set root_key = "" -%} {%- endif -%} {%- if root_key and nav.routes[root_key] is defined -%} {%- for group_key, group in nav.groups -%} {%- set_global group_routes = [] -%} {%- if group_key == "start" -%} {%- set_global group_routes = group_routes | concat(with=[root_key]) -%} {%- endif -%} {%- if group_key == "cross-version" and nav.routes["/versions/"] is defined -%} {%- set_global group_routes = group_routes | concat(with=["/versions/"]) -%} {%- endif -%} {%- for key, node in nav.routes -%} {%- if node.parent == root_key and node.group == group_key -%} {%- set_global group_routes = group_routes | concat(with=[key]) -%} {%- endif -%} {%- endfor -%} {%- set_global unique_group_routes = [] -%} {%- for candidate_route in group_routes -%} {%- if not (emitted_routes is containing(candidate_route)) -%} {%- set_global unique_group_routes = unique_group_routes | concat(with=[candidate_route]) -%} {%- set_global emitted_routes = emitted_routes | concat(with=[candidate_route]) -%} {%- endif -%} {%- endfor -%} {%- if unique_group_routes | length > 0 -%} {%- set_global group_active = false -%} {%- for route_key in group_routes -%} {%- if mapped_url == route_key -%}{%- set_global group_active = true -%}{%- endif -%} {%- endfor -%}
